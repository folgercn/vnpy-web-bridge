"""Offline structural contract audit; no execution/admission or synthetic evidence."""
import copy
from datetime import datetime
import json
from pathlib import Path

from jsonschema import Draft202012Validator, FormatChecker

ROOT = Path(__file__).resolve().parents[3]
DIRECTORY = ROOT / 'docs/research-lab/specs'
schema = json.loads((ROOT / 'docs/schemas/research-experiment-spec-v2.schema.json').read_text())
Draft202012Validator.check_schema(schema)
formats = FormatChecker()


@formats.checks('date-time', raises=ValueError)
def utc_micros(value):
    if not isinstance(value, str):
        return True  # type assertion belongs to Schema
    datetime.strptime(value, '%Y-%m-%dT%H:%M:%S.%fZ')
    return True


validator = Draft202012Validator(schema, format_checker=formats)
count = 0


def check(name, value, expected):
    global count
    errors = list(validator.iter_errors(value))
    assert bool(errors) != expected, (name, [e.message for e in errors])
    count += 1
    print(('ACCEPT ' if expected else 'REJECT ') + name)


examples = {}
for file in sorted(DIRECTORY.glob('spec-*-example.json')):
    value = json.loads(file.read_text())
    examples[value['experiment_type']] = value
    check(file.name, value, True)
for file in sorted(DIRECTORY.glob('reject-*.json')):
    check(file.name, json.loads(file.read_text()), False)


def mutation(name, kind, mutate, expected=False):
    value = copy.deepcopy(examples[kind])
    mutate(value)
    check(name, value, expected)
