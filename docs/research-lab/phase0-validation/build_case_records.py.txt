def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def canon(v):
 if v is None:return 'null'
 if isinstance(v,bool):return 'true' if v else 'false'
 if isinstance(v,int):return str(v)
 if isinstance(v,str):return '"'+''.join('\\"' if x=='"' else '\\\\' if x=='\\' else '\\u%04x'%ord(x) if ord(x)<32 else x for x in v)+'"'
 if isinstance(v,list):return '['+','.join(canon(x) for x in v)+']'
 if isinstance(v,dict):return '{'+','.join(canon(k)+':'+canon(v[k]) for k in sorted(v))+'}'
 raise ValueError(type(v))
def digest(v):return hashlib.sha256(canon(v).encode()).hexdigest()
def decimals(v):
 if isinstance(v,float):return '0' if v==0 else format(decimal.Decimal(str(v)).normalize(),'f')
 if isinstance(v,dict):return {k:decimals(x) for k,x in v.items()}
 if isinstance(v,list):return [decimals(x) for x in v]
 return v
def load(p):return decimals(json.loads(p.read_text()))
